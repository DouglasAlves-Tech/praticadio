def _safe_date(y: int, m: int, d: int) -> str:
    tentar:
        retornar dt.date(y, m, d).strftime("%Y-%m-%d")
    exceto (ValueError, TypeError):
        retornar INVÁLIDO


def sanitize_date(value: Any) -> str:
    Se o valor for None ou for igual a "":
        retornar INVÁLIDO

    se isinstance(value, dt.datetime):
        valor de retorno.strftime("%Y-%m-%d")
    se isinstance(value, dt.date):
        valor de retorno.strftime("%Y-%m-%d")

    s = str(valor).strip()
    se não s:
        retornar INVÁLIDO

    # AAAA-MM-DD / AAAA/MM/DD / AAAA.MM.DD
    m = re.match(r"^(\d{4})[-/.](\d{1,2})[-/.](\d{1,2})$", s)
    se m:
        retornar _safe_date(int(m.group(1)), int(m.group(2)), int(m.group(3)))

    # MM/DD/AAAA ou MM-DD-AAAA
    m = re.match(r"^(\d{1,2})[-/](\d{1,2})[-/](\d{4})$", s)
    se m:
        retornar _safe_date(int(m.group(3)), int(m.group(1)), int(m.group(2)))

    # MM/DD/AA ou MM-DD-AA
    m = re.match(r"^(\d{1,2})[-/](\d{1,2})[-/](\d{2})$", s)
    se m:
        yy = int(m.group(3))
        ano = 2000 + yy se yy < 70, caso contrário, 1900 + yy
        retornar _safe_date(ano, int(m.group(1)), int(m.group(2)))

    # Mês DD, AAAA / Seg DD, AAAA / Mês DD, AAAA
    m = re.match(
        r"^([A-Za-z]+)\s+(\d{1,2})(?:st|nd|rd|th)?[,\s]+(\d{4})$",
        s,
    )
    se m:
        mon_key = m.group(1).lower().rstrip(".")
        se mon_key estiver em MESES:
            retornar _safe_date(int(m.group(3)), MONTHS[mon_key], int(m.group(2)))

    retornar INVÁLIDO

def _words_to_int(text: str) -> int | None:
    "Converter 'vinte e vinte e quatro', 'dois mil e vinte e dois', 'oitenta e dois mil', etc."
    tokens = re.findall(r"[az]+", text.lower())
    se não houver tokens:
        retornar Nenhum
    se algum(t não estiver em WORDS_TO_NUM para t em tokens):
        retornar Nenhum

    # Trata caso "vinte e vinte e quatro" (estilo ano: pares de bolsas)
    # quando todos os tokens são <= 90 e nenhum é cem/mil
    se todos(WORDS_TO_NUM[t] < 100 para t em tokens):
        # tenta ler como dois pares: "vinte e vinte e quatro" -> 20*100 + 24
        nums = [PALAVRAS_PARA_NUM[t] para t em tokens]
        #combinar pares: ex [20, 20, 4] -> dois números: 20 e 24
        grupos: lista[int] = []
        i = 0
        enquanto i < len(nums):
            se i + 1 < len(nums) e nums[i] >= 20 e nums[i] % 10 == 0 e nums[i + 1] < 10:
                grupos.append(nums[i] + nums[i + 1])
                i += 2
            outro:
                grupos.append(nums[i])
                i += 1
        se len(grupos) == 2:
            retornar grupos[0] * 100 + grupos[1]
        se len(grupos) == 1:
            retornar grupos[0]

    # Estilo "dois mil e vinte e dois" / "oitenta e dois mil"
    total = 0
    atual = 0
    para t em tokens:
        v = PALAVRAS_PARA_NUM[t]
        se v == 100:
            atual = máx(1, atual) * 100
        senão se v == 1000:
            atual = máx(1, atual) * 1000
            total += atual
            atual = 0
        outro:
            corrente += v
    retornar total + atual

    cabeçalho_saída: lista[str] = []
    para src, san_name, _ em COLUMN_PIPELINE:
        out_header.append(src)
        se san_name:
            out_header.append(san_name)
    ws_out.append(out_header)

    # Estilo do cabeçalho
    fonte_do_cabeçalho = Font(negrito=True, cor="FFFFFF")
    header_fill = PatternFill("solid", fgColor="1F2937")
    sanitized_fill = PatternFill("solid", fgColor="065F46")
    para col_idx, nome em enumerate(out_header, start=1):
        célula = ws_out.cell(linha=1, coluna=col_idx)
        cell.font = fonte_do_cabeçalho
        cell.fill = sanitized_fill se name.endswith("Sanitized") senão header_fill

    # Estatísticas de inválidos
    invalid_counts: dict[str, int] = {}

    para r em linhas[1:]:
        out_row: list[Any] = []
        para src, san_name, fn em COLUMN_PIPELINE:
            raw = r[nome_para_idx[src]]
            out_row.append(raw)
            Se san_name e fn não forem None:
                limpar = fn(bruto)
                se limpo == INVÁLIDO:
                    invalid_counts[san_name] = invalid_counts.get(san_name, 0) + 1
                out_row.append(limpo)
        ws_out.append(out_row)

    # Largura das colunas
    para col_idx em range(1, len(out_header) + 1):
        letra = obter_letra_da_coluna(idx_coluna)
        ws_out.column_dimensions[letter].width = max(14, len(out_header[col_idx - 1]) + 2)
    ws_out.freeze_panes = "A2"

    wb_out.save(caminho_de_saída)
    retornar contagens_inválidas
